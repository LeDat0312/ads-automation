## 📝 CODE EXAMPLE

### **FastAPI + OpenAI Integration:**

```python
# app/services/ai_service.py
from openai import OpenAI
from google import generativeai as genai
import os

class AIService:
    def __init__(self):
        self.openai_client = OpenAI(api_key=os.getenv("OPENAI_API_KEY"))
        genai.configure(api_key=os.getenv("GEMINI_API_KEY"))
        self.gemini_model = genai.GenerativeModel('gemini-pro')
    
    async def get_insights(self, data: dict) -> str:
        """Get AI insights from data"""
        prompt = f"""
        Analyze this Facebook Ads data and provide insights:
        {data}
        
        Provide:
        1. Key insights
        2. Recommendations
        3. Alerts
        """
        
        # Use ChatGPT
        response = self.openai_client.chat.completions.create(
            model="gpt-4",
            messages=[
                {"role": "system", "content": "You are a Facebook Ads expert."},
                {"role": "user", "content": prompt}
            ]
        )
        
        return response.choices[0].message.content
    
    async def optimize_budget(self, adset_data: dict) -> dict:
        """AI-powered budget optimization"""
        prompt = f"""
        Optimize budget for this adset:
        {adset_data}
        
        Provide:
        1. Recommended budget
        2. Reasoning
        3. Expected ROI
        """
        
        # Use Gemini
        response = self.gemini_model.generate_content(prompt)
        
        return {
            "recommended_budget": response.text,
            "reasoning": response.text,
            "confidence": 0.85
        }
```

### **FastAPI Routes:**

```python
# app/api/routes/dashboard.py
from fastapi import APIRouter, Depends
from app.services.dashboard import DashboardService
from app.services.ai_service import AIService
from app.schemas.dashboard import DashboardResponse

router = APIRouter()

@router.get("/dashboard", response_model=DashboardResponse)
async def get_dashboard(
    dashboard_service: DashboardService = Depends(),
    ai_service: AIService = Depends()
):
    """Get dashboard overview with AI insights"""
    # Get data
    data = await dashboard_service.get_dashboard_data()
    
    # Get AI insights
    insights = await ai_service.get_insights(data)
    
    return {
        "data": data,
        "ai_insights": insights
    }
```
