# Code pull và fix lỗi 502 - Phiên bản đúng

cd ~/ads-automation
git pull origin main

# 1. Kiểm tra supervisor config để biết tên service đúng
sudo cat /etc/supervisor/conf.d/ads-automation.conf

# 2. Kiểm tra port 8000 (dùng ss thay vì netstat)
sudo ss -tlnp | grep 8000

# 3. Xem logs của service ads-automation
sudo tail -50 /var/log/supervisor/ads-automation-stdout.log
sudo tail -50 /var/log/supervisor/ads-automation-stderr.log

# 4. Kiểm tra process đang chạy
ps aux | grep uvicorn | grep -v grep

# 5. Kiểm tra syntax Python
python3 -m py_compile app/api/routes/dashboard.py
python3 -m py_compile app/main.py

# 6. Restart service (tên đúng là ads-automation)
sudo supervisorctl stop ads-automation
sleep 2
sudo supervisorctl start ads-automation
sleep 3

# 7. Kiểm tra lại
sudo supervisorctl status
sudo ss -tlnp | grep 8000

# 8. Reload nginx
sudo nginx -t
sudo systemctl reload nginx

# Nếu vẫn lỗi, kiểm tra supervisor config:
sudo cat /etc/supervisor/conf.d/ads-automation.conf

# Và kiểm tra xem command có đúng không:
# Command phải là: uvicorn app.main:app --host 0.0.0.0 --port 8000
