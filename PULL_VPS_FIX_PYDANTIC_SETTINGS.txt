# Code fix lỗi ModuleNotFoundError: No module named 'pydantic_settings'

cd /var/www/ads-automation

# 1. Activate virtual environment
source venv/bin/activate

# 2. Cài đặt pydantic-settings
pip install pydantic-settings

# 3. Nếu có requirements.txt, cài đặt tất cả dependencies
if [ -f "requirements.txt" ]; then
    pip install -r requirements.txt
else
    # Cài đặt các package cần thiết
    pip install pydantic-settings fastapi uvicorn sqlalchemy psycopg2-binary python-dotenv
fi

# 4. Kiểm tra import
python3 -c "from pydantic_settings import BaseSettings; print('✅ OK')"
python3 -c "from app.main import app; print('✅ App OK')"

# 5. Restart service
sudo supervisorctl stop ads-automation
sleep 2
sudo supervisorctl start ads-automation
sleep 5

# 6. Kiểm tra
sudo supervisorctl status
sudo ss -tlnp | grep 8000
curl http://localhost:8000/health

# 7. Xem logs nếu vẫn lỗi
sudo tail -30 /var/log/ads-automation.log
