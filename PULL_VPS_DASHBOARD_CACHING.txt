# Script để pull code Dashboard với Caching lên VPS
# Chạy các lệnh sau trên VPS:

cd ~/ads-automation
git stash
git pull origin main
python3 -m py_compile app/api/routes/dashboard.py app/main.py
sudo supervisorctl restart all
sudo supervisorctl status
