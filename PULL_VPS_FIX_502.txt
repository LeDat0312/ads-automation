# Lệnh pull và fix lỗi 502 Bad Gateway trên VPS Ubuntu

# Cách 1: Chạy script tự động (khuyến nghị)
cd ~/ads-automation
bash CHECK_VPS_502_ERROR.sh
bash FIX_VPS_502_ERROR.sh

# Cách 2: Chạy từng lệnh thủ công

# 1. Pull code mới nhất
cd ~/ads-automation
git pull origin main

# 2. Kiểm tra syntax
python3 -m py_compile app/api/routes/dashboard.py
python3 -m py_compile app/main.py

# 3. Kiểm tra trạng thái service
sudo supervisorctl status

# 4. Xem logs lỗi
sudo tail -50 /var/log/supervisor/ads-automation-api-stderr.log
sudo tail -50 /var/log/nginx/error.log

# 5. Restart service
sudo supervisorctl restart ads-automation-api

# 6. Kiểm tra lại
sudo supervisorctl status
sudo netstat -tlnp | grep 8000

# 7. Reload nginx
sudo nginx -t
sudo systemctl reload nginx
