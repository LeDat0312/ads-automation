# Lệnh fix 502 Bad Gateway trên VPS

# 1. Kiểm tra syntax error
cd ~/ads-automation
python3 -m py_compile app/api/routes/dashboard.py app/services/facebook_api.py

# 2. Kiểm tra supervisor status
sudo supervisorctl status

# 3. Xem logs để tìm lỗi
sudo tail -100 /var/log/ads-automation.log

# 4. Restart service
sudo supervisorctl restart ads-automation
sleep 3
sudo supervisorctl status

# 5. Kiểm tra port
sudo netstat -tlnp | grep 8000 || sudo ss -tlnp | grep 8000

# 6. Nếu vẫn lỗi, kiểm tra nginx
sudo nginx -t
sudo tail -20 /var/log/nginx/error.log
