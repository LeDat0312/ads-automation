=============================================================================
TROUBLESHOOTING - XỬ LÝ LỖI
=============================================================================

LỖI: "Service failed to start"
GIẢI PHÁP:
  sudo journalctl -u metaupdate-backend -n 50
  # Xem logs để biết lỗi cụ thể

LỖI: "Port already in use"  
GIẢI PHÁP:
  sudo lsof -i :8000
  sudo kill -9 <PID>
  sudo systemctl start metaupdate-backend

LỖI: "Database connection failed"
GIẢI PHÁP:
  sudo systemctl status postgresql
  sudo systemctl start postgresql
  # Kiểm tra database credentials trong .env

LỖI: "Frontend 404 Not Found"
GIẢI PHÁP:
  cd ~/ads-automation/frontend
  npm run build
  # Đảm bảo dist/ folder được tạo

LỖI: "Permission denied"
GIẢI PHÁP:
  sudo chown -R $USER:$USER ~/ads-automation
  chmod +x VPS_PULL_LATEST.sh
