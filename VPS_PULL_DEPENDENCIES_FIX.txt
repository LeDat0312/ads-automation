# Bước 2: Chuyển đến thư mục project

git pull origin main

# Bước 4: Cấp quyền thực thi cho script
chmod +x VPS_PULL_DEPENDENCIES_FIX.sh

# Bước 5: Chạy script
./VPS_PULL_DEPENDENCIES_FIX.sh

# ============================================
# CÁCH 2: Chạy từng lệnh thủ công
# ============================================

# Bước 1: SSH vào VPS
ssh adsuser@your-vps-ip

# Bước 2: Chuyển đến thư mục project
cd /home/adsuser/ads-automation

# Bước 3: Stash local changes (nếu có)
git stash

# Bước 4: Pull code mới nhất
git pull origin main

# Bước 5: Cài đặt frontend dependencies
cd frontend
npm install

# Bước 6: Build frontend
npm run build

# Bước 7: Quay lại thư mục root
cd ..

# Bước 8: Kích hoạt virtual environment
source venv/bin/activate

# Bước 9: Chạy migrations
python3 -m migrations.add_last_error_to_facebook_accounts
python3 -m migrations.add_color_hex_to_channel_groups

# Bước 10: Deactivate venv
deactivate

# Bước 11: Restart backend service
sudo systemctl restart ads-automation

# Hoặc nếu dùng supervisor:
sudo supervisorctl restart backend

# Bước 12: Kiểm tra status
sudo systemctl status ads-automation

# Bước 13: Xem logs (nếu cần)
sudo journalctl -u ads-automation -f

# 3. Kiểm tra database migrations
# (Trong Python shell)
source venv/bin/activate
python3 << EOF
from app.core.database import init_db, get_db_session
from sqlalchemy import text

init_db()
db = get_db_session()

# Check last_error column
result = db.execute(text("""
    SELECT column_name 
    FROM information_schema.columns 
    WHERE table_name='facebook_accounts' 
    AND column_name='last_error'
""")).fetchone()
print(f"✅ last_error column exists: {result is not None}")

# Check color_hex column
result = db.execute(text("""
    SELECT column_name 
    FROM information_schema.columns 
    WHERE table_name='channel_groups' 
    AND column_name='color_hex'
""")).fetchone()
print(f"✅ color_hex column exists: {result is not None}")

db.close()
EOF
deactivate
