# Code debug port 8000 không chạy

cd ~/ads-automation

# 1. Xem logs thực tế (theo config supervisor)
sudo tail -50 /var/log/ads-automation.log

# 2. Kiểm tra port 8000
sudo ss -tlnp | grep 8000
sudo lsof -i :8000

# 3. Test kết nối trực tiếp
curl -v http://localhost:8000/health

# 4. Kiểm tra process
ps aux | grep uvicorn | grep -v grep

# 5. Restart và xem logs real-time
sudo supervisorctl stop ads-automation
sleep 2
sudo tail -f /var/log/ads-automation.log &
TAIL_PID=$!
sudo supervisorctl start ads-automation
sleep 5
kill $TAIL_PID

# 6. Kiểm tra lại port
sudo ss -tlnp | grep 8000

# 7. Nếu vẫn không có port, thử chạy thủ công để xem lỗi
cd /var/www/ads-automation
source venv/bin/activate
python3 -c "from app.main import app; print('Import OK')"
uvicorn app.main:app --host 0.0.0.0 --port 8000
